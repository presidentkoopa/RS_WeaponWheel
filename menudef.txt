// Its own page, not an addition to the engine's.
//
// The native VR wheel has a full options page of its own under vr_wheel_*, and
// this is a different system rather than a skin on that one -- different
// layout, different selection, different equip path. Sharing a page would mean
// two sets of sliders that look interchangeable and are not.
//
// Added to BOTH option menus: OptionsMenuSimple is what a lot of people
// actually browse, and a mod that only appears in the full one is a mod half
// its users never find.
AddOptionMenu OptionsMenu       { Submenu "Wrist Rig", wr_Options }
AddOptionMenu OptionsMenuSimple { Submenu "Wrist Rig", wr_Options }

OptionMenu wr_Options
{
	Title "WRIST RIG"

	StaticText ""
	StaticText "Behaviour", gold
	Option "Open at level start",       "wr_autoopen",     "OnOff"
	Slider "Seconds before it folds",   "wr_locktics",     35, 700, 35, 0

	StaticText ""
	StaticText "Size", gold
	Slider "Overall scale",             "wr_scale",        0.3, 3.0, 0.05, 2

	StaticText ""
	StaticText "Placement", gold
	// Every one of these is live -- open the rig and drag a slider and the next
	// tic has it. That is the whole reason they are sliders and not constants.
	Slider "Ring distance ahead",       "wr_forward",      5.0, 120.0, 1.0, 0
	Slider "Ring radius",               "wr_radius",       1.0, 40.0, 0.5, 1
	Slider "Height above hand",         "wr_rise",        -8.0, 12.0, 0.5, 1
	Slider "Lean toward you",           "wr_tilt",       -30.0, 30.0, 1.0, 0
	// Off by default: a tracked controller reports ninety degrees of roll just
	// from how you hold it, so raw wrist roll swings the ring behind you.
	Slider "Wrist roll spins the ring", "wr_roll",         0.0, 1.0, 0.05, 2

	StaticText ""
	StaticText "Cards", gold
	Slider "Card width",                "wr_panel_w",      1.0, 12.0, 0.25, 2
	Slider "Card height",               "wr_panel_h",      1.0, 12.0, 0.25, 2
	Slider "Hovered card steps out",    "wr_pop",          0.0, 6.0, 0.25, 2
	Slider "Hovered card pulse",        "wr_pulse",        0.0, 0.5, 0.01, 2
	// 0 makes the ring appear finished, which is how this behaved before the
	// cards grew out of the wrist.
	Slider "Cards grow in over (tics)", "wr_growtics",     0, 20, 1, 0
	Option "Ammo counts and gauges",    "wr_ammo",         "OnOff"
	// Off falls back to the sampled plate: cheaper, blurs up close, and cannot
	// glow -- a halo needs a distance field to read past the shape's edge.
	Option "Sharp plates (SDF)",        "wr_sdf",          "OnOff"
	Slider "Corner rounding",           "wr_plate_radius", 0, 15, 1, 0
	Slider "Plate border",              "wr_plate_border", 0, 15, 1, 0
	// The other way to build a card: artwork painted into a texture in 2D pixel
	// coordinates instead of assembled from billboards in map units. Both work.
	// A raster, so it shows its pixels up close where the field payloads do not.
	Option "Painted card faces",        "wr_canvas",       "OnOff"

	StaticText ""
	StaticText "The room", gold
	// All three borrow a single global engine slot each. A co-loaded mod using
	// the same slot gets it back the moment the rig closes.
	Option "Laser has dust in it",      "wr_vbeam",        "OnOff"
	Slider "Opening light sweep (tics)","wr_sweep",        0, 40, 1, 0
	Slider "Mist density",              "wr_fog",          0.0, 1.0, 0.05, 2
	// A mark under your feet, repeated around a circle and spinning. 0 is off.
	Slider "Floor marks (size)",        "wr_floor",        0.0, 40.0, 1.0, 0
	Slider "Floor mark count",          "wr_floor_count",  1.0, 24.0, 1.0, 0
	Slider "Floor mark spin",           "wr_floor_spin", -60.0, 60.0, 2.0, 0

	StaticText ""
	StaticText "Feel", gold
	// The rig is worn where you cannot comfortably look at it, so the ear and
	// the wrist do most of the work of saying what just happened.
	Option "Sound",                     "wr_sound",        "OnOff"
	Slider "Volume",                    "wr_volume",       0.0, 1.0, 0.05, 2
	// vr_enable_haptics still overrules this, so turning rumble off in the
	// engine cannot be undone from here.
	Slider "Controller rumble",         "wr_haptics",      0.0, 2.0, 0.1, 1

	StaticText ""
	StaticText "Reach", gold
	// Depth tolerance more than reach: how far in front of or behind the grid
	// your hand may sit and still register.
	// 0 turns reaching off and leaves the beam as the only way in. Drop
	// wr_forward toward 0 to bring the ring into actual arm reach.
	Slider "Reach-in radius (0 = off)", "wr_touch",        0.0, 20.0, 0.5, 1
	Slider "Stick deadzone",            "wr_stickdead",    500, 10000, 250, 0

	StaticText ""
	StaticText "Integration", gold
	Option "Slow time while open (BulletTimeX)", "wr_bullettime", "OnOff"

	StaticText ""
	StaticText "Controls", gold
	// Whichever hand you summon it on is the hand that wears it, points at it,
	// and receives what you pick -- so the bind already says which hand you
	// meant and there is no handedness setting to get wrong.
	Control "Off hand wheel",           "wr_toggle_off"
	Control "Main hand wheel",          "wr_toggle_main"

	StaticText ""
	StaticText "Troubleshooting", gold
	Option "Print diagnostics on open", "wr_debug",        "OnOff"

	StaticText ""
}

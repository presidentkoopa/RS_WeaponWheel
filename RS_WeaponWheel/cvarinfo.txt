// All live -- change them with the rig open and the next tic picks them up.
//
// SCALE: vr_vunits_per_meter is 34, so one map unit is about 3cm. Every default
// below is written as a real-world distance first and converted, because the
// first pass at this used flat-screen numbers and put the cards a metre off
// each side of the hand.

// Panel face size. 3.5 x 2.5 units is about 10cm x 7cm -- a playing card.
user float wr_panel_w  = 3.5;
user float wr_panel_h  = 2.5;

// How far the panels orbit out from the hand, and how high above it they ride.
// 7 units is about 20cm: far enough to read, close enough to reach without
// moving your arm.
user float wr_radius   = 5.0;
user float wr_rise     = 2.0;

// Leans the tops toward you.
user float wr_tilt     = 12.0;

// Touch radius, and it is DEPTH tolerance more than reach. The grid is a flat
// plane; this is how far your hand may sit in front of or behind it and still
// count. Small values mean a perfectly aimed hand misses because it drifted a
// few centimetres forward. Nearest-hit resolves any overlap, so generous is
// free. 7 units is about 20cm.
user float wr_touch    = 7.0;

// Open the rig automatically when a level starts. On while iterating; turn it
// off once it is bound to something you actually want to press.
user bool  wr_autoopen = true;

// How much wrist roll spins the fan. 0 ignores it, 1 follows it exactly.
user float wr_roll     = 0.0;

// Which generation of the defaults above this config has seen.
//
// user cvars are ARCHIVED: once your config holds wr_radius 30, editing the
// default here can never reach you again -- the saved value wins forever. So
// the mod stamps a version, and when it ships new geometry it rewrites the
// values once and bumps the stamp. Your own tuning survives everything after
// that, because the stamp only moves when the mod's numbers change.
user int   wr_cfgver   = 0;


// How far the hovered card steps toward your eye, in map units.
user float wr_pop      = 1.5;

// Tics the rig stays up with nothing hovered before folding away. 35 = 1 second.
// Hovering a card resets it.
user int   wr_locktics = 140;

// Master size. Grows distance, card size and reach together, so "make it
// bigger" stays one thought instead of four sliders.
user float wr_scale    = 1.0;



// Slow time while the wheel is up, via BulletTimeX if it is loaded. Its
// activate is a toggle, so turning this on means opening the wheel during a
// bullet time you started yourself will cancel it.
user bool  wr_bullettime = true;

// How far the stick must leave centre before it picks a slot. UserCmd units:
// a walk is about 6400.
user float wr_stickdead  = 3000.0;


// How much the hovered card breathes. 0 holds it still.
user float wr_pulse      = 0.10;

// How far in front of your hand the ring floats, along where you were aiming
// when it opened. 34 units is about a metre. This is the number that decides
// how far you have to turn to cross the ring: further away means a smaller
// sweep, because the whole thing subtends less.
user float wr_forward  = 34.0;


// FEEL
//
// The rig is worn where you cannot comfortably look at it, so the ear and the
// wrist do most of the work of telling you what is happening. These are the
// dials for that.

// Menu clicks on hover, expand, confirm and dismiss. The rig ships its own --
// see sndinfo.txt.
user bool  wr_sound    = true;
user float wr_volume   = 0.65;

// Controller rumble, as a multiplier on what the rig asks for. 0 is silent
// hands. vr_enable_haptics still overrules this, so turning haptics off in the
// engine cannot be undone by a mod.
user float wr_haptics  = 1.0;

// Tics the cards take to grow out of your wrist to the ring. 0 makes them
// appear finished, which is how this used to behave.
user float wr_growtics = 6.0;

// Print the ammo each weapon has left under its name, and rest a card that has
// none in a colour that says so.
user bool  wr_ammo     = true;

// Diagnostics on the console when the rig opens. Off: these fired on every
// single summon, and a line of console for something that worked is noise.
user bool  wr_debug    = false;



// THE ROOM
//
// These three are SINGLE GLOBAL SLOTS in the engine with no way to read them
// back, so while the rig holds one, a co-loaded mod using the same slot loses
// it. That is survivable because the engine's contract for all three is
// "publish it every tic while it is on" -- so anything else using them
// re-publishes and recovers by itself the moment the rig lets go. Each is
// separately switchable anyway.

// The laser gets air in it: a raymarched cone laid along the same ray the
// engine laser already points down, with dust drifting through it. 0 is off.
user bool  wr_vbeam         = true;
user float wr_vbeam_inner   = 1.2;    // hot core radius
user float wr_vbeam_outer   = 5.0;    // where it has faded to nothing
user float wr_vbeam_density = 0.55;
user float wr_vbeam_dust    = 0.5;

// A ring of light washing outward across the room when the rig opens, tested
// per pixel against world position so it crosses floor, wall and ceiling as one
// unbroken line. Tics it takes; 0 is off.
user float wr_sweep         = 14.0;
user float wr_sweep_reach   = 340.0;  // how far it travels, map units
user float wr_sweep_bright  = 1.4;

// Mist for the cards to sit in, tinted to whatever slot you are hovering.
// OFF by default, and it is the rude one: a map may have authored its own fog
// slab, and this takes it for as long as the rig is up.
user float wr_fog           = 0.0;    // density; 0 is off
user float wr_fog_rise      = 6.0;    // how far above the ring its surface sits
user float wr_fog_scatter   = 0.8;    // lets the beam light the mist


// THE CARDS THEMSELVES
//
// Two ways to draw a plate. BB_PANEL stretches one small rounded-rect texture
// -- cheap, and it blurs when a card is held close. BB_SDFPANEL solves the same
// rectangle per pixel: crisp at any size, and it can take a halo, which a
// sampled plate structurally cannot. Off gives you the old look back.
user bool wr_sdf           = true;

// Only the solved plate reads these. 0-15 each, across the half-extent.
user int  wr_plate_radius  = 4;    // corner rounding
user int  wr_plate_border  = 2;    // rim width; 0 is a plain filled plate


// THE FLOOR
//
// A mark under your feet, repeated around a circle and spinning. One shape
// slot: the repeat folds the coordinate rather than drawing copies, so eight
// marks and eight hundred cost the same. Size in map units; 0 is off.
user float wr_floor        = 0.0;
user float wr_floor_kind   = 5.0;   // 1 disc 2 ring 3 square 4 outline 5 cross 6 hex 7 triangle
user float wr_floor_count  = 8.0;   // copies around the circle
user float wr_floor_space  = 46.0;  // radius of that circle
user float wr_floor_spin   = 14.0;  // degrees per second
user float wr_floor_thick  = 0.18;
user float wr_floor_bright = 1.2;
// How far each mark splits open down its middle as the rig arrives.
user float wr_floor_seam   = 0.5;


// CANVAS FACES
//
// The other way to build a card. Instead of one billboard per element, each
// placed in map units every tic, the artwork is PAINTED into a texture in
// ordinary 2D pixel coordinates and the card shows that one image.
//
// Both work and neither is strictly better. A canvas is a raster, so held
// close in VR it shows its pixels where the distance-field payloads never
// will -- which is why the name and the count stay as field billboards on top
// either way, and only the artwork underneath them changes.
//
// Twelve canvas textures are declared in animdefs.txt. A card past that falls
// back to the composed face on its own, so a big weapon set degrades rather
// than breaking. Painted once on open; nothing on a card changes while the
// ring is up.
user bool wr_canvas = false;

// How far the weapon sprite runs off the card, as a multiple of card width.
// 1.0 fits it inside; above that it crops, which is the thing a billboard
// cannot do at all -- a quad draws its whole texture, so the composed card can
// only ever shrink a sprite until it fits.
user float wr_canvas_bleed  = 1.30;

// A dark offset twin under the sprite. Weapon art is lit every which way
// across a weapon set; a shadow gives all of it the same weight.
user bool  wr_canvas_shadow = true;

// One dimmed row every three, over the whole face. 0 is off.
user float wr_canvas_scan   = 0.22;

// Halo strength on the card you are pointing at, as a multiplier. This is the
// one thing the distance-field payloads can do that a sampled texture cannot --
// a glow is placed by reading the field OUTSIDE the shape, and a texture has
// nothing out there to read. It only ever fires on the hovered card, which is
// why it is easy to miss entirely. Crank it to see what SDF is for.
user float wr_glow = 1.6;


// SPARKS
//
// A burst off the card you just took. 0 is off.
user float wr_sparks       = 18.0;   // particles per commit
user float wr_sparks_speed = 1.4;
// 0 the slot's colour, 1 white hot, 2 rainbow.
//
// Slot colour is the default because it is doing a JOB -- same hue as the card,
// the accent and the beam, so the burst confirms which one you hit. Rainbow is
// prettier and says nothing.
user int   wr_spark_color  = 0;

// Degrees the taken card spins as the ring folds away. 0 holds it still.
user float wr_flip         = 360.0;


// GLOW WAVE
//
// Ripples the glows a map ALREADY HAS, per pixel, anchored at the ring -- with
// separate phases for wall top, wall bottom, floor and ceiling so the wave
// climbs a room instead of flashing it.
//
// In a map with no glowing surfaces this does nothing at all. Its whole effect
// depends on the level you are standing in, which is why it is off by default
// and not broken when it looks dead. 0 is off.
user float wr_wave       = 0.0;
user float wr_wave_len   = 120.0;
user float wr_wave_speed = 1.6;
user float wr_wave_sharp = 1.4;


// CARDS, ONE MORE PASS

// How far the cards you are NOT pointing at fade back. 1 is no dimming.
// Only applies once something is hovered -- fading the whole ring the moment it
// opens would look like it failed to load.
user float wr_dim = 0.55;

// One dynamic light, on the hovered card, in that slot's colour, breathing on
// the same pulse the card does. ONE, not one per card: lights cost per light
// per surface they touch, and only one card is ever the answer to "which".
user bool  wr_light      = true;
user float wr_light_size = 44.0;

// A corner mark on a weapon one of your hands already holds. Informational
// only -- the card stays selectable, because picking your other hand's gun
// swaps them or finds a free clone, which is a reasonable thing to ask for.
user bool  wr_marker = true;

// Which payload draws the ammo bezel.
//   0 LED segment, 1 LCD (punched out of a lit plate), 2 lozenge badge
// Pick by which reads better against the room, not which is more correct.
user int   wr_readout = 0;

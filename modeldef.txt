// The hand the rig shows while the cards are up.
//
// Written to match the MODELDEFs in RS_Main and RS_Lance that are known to
// work in this engine, rather than to the wiki: explicit Offset, mirrored X
// scale, and a sprite that actually exists as a lump.
//
// Bound to WR_Hand, a carrier class, NOT to Doom's Fist -- binding it to Fist
// would give every game's fist a model whether or not this menu was opened.
//
// PUNG is the vanilla fist sprite. RS_Lance's own comment makes the rule
// explicit: "the sprites themselves still have to exist as lumps even though
// the model replaces them." PUNG is in every IWAD, so nothing ships here.
//
// Model choice is deliberate: doomknuckles is ONE frame and ONE surface. The
// other candidates are 57- and 75-frame multi-surface scene exports, and a
// second surface with no Skin 1 is a second surface that does not draw.
Model WR_Hand
{
	Path "models"
	Model 0 "wr_hand.md3"
	Skin 0 "wr_hand.png"
	// Half size, in the MODELDEF rather than at runtime -- one number, applied
	// before anything else can fight it.
	Scale -0.5 0.5 0.5
	Offset 0.0 0.0 0.0

	FrameIndex PUNG A 0 0
}
